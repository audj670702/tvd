<script>
'use strict';
const VERSION='0.3.0';
const $=id=>document.getElementById(id);
let canales=[],eos=[],ocupado=true,conectado=false,recibido=false,intentos=0;
const desplegados=new Set(),filtrosEO=new Map();
const enviar=(tipo,extra={})=>parent.postMessage({tipo,version:VERSION,...extra},'*');
const nombreEO=contexto=>eos.find(e=>e.codigoContexto===contexto);
function render(){
  const filtro=$('buscar').value.trim().toLocaleLowerCase('es');
  const visibles=canales.filter(c=>[c.codigo,c.nombre,c.tipo,c.enteOperador,nombreEO(c.enteOperador)?.nombre,...(c.eoCompartidas||[]).map(e=>nombreEO(e)?.nombre||e)].join(' ').toLocaleLowerCase('es').includes(filtro));
  $('filas').replaceChildren();
  if(!visibles.length){const tr=document.createElement('tr'),td=document.createElement('td');td.colSpan=9;td.className='empty';td.textContent=recibido?(canales.length?'Sin coincidencias.':'No hay canales registrados.'):'Esperando catálogo…';tr.append(td);$('filas').append(tr);}
  visibles.forEach((c,i)=>{
    const tr=document.createElement('tr');
    for(const [valor,clase] of [[String(i+1),'num'],[c.codigo,'code'],[c.nombre,''],[c.tipo,'']]){const td=document.createElement('td');td.textContent=valor||'';td.className=clase;tr.append(td);}
    const estado=document.createElement('td'),badge=document.createElement('span');badge.className='badge'+(c.activo?'':' inactive');badge.textContent=c.activo?'Activo':'Inactivo';estado.append(badge);tr.append(estado);
    const td=document.createElement('td'),select=document.createElement('select');select.setAttribute('aria-label',`Empresa de ${c.codigo}`);
    const agregar=(valor,texto)=>{const option=document.createElement('option');option.value=valor;option.textContent=texto;select.append(option);};
    agregar('','Sin asignar');
    eos.forEach(e=>agregar(e.codigoContexto,`${e.codigoEO} · ${e.nombre}`));
    if(c.enteOperador&&!nombreEO(c.enteOperador))agregar(c.enteOperador,`${c.enteOperador} · no disponible`);
    select.value=c.enteOperador||'';select.disabled=ocupado||!conectado;
    select.addEventListener('change',()=>guardar(c,{enteOperador:select.value,eoCompartidas:(c.eoCompartidas||[]).filter(e=>e!==select.value)}));
    td.append(select);tr.append(td);
    const compartidas=document.createElement('td'),detalle=document.createElement('details'),resumen=document.createElement('summary');
    const ids=c.eoCompartidas||[];
    resumen.textContent=ids.length?`${ids.length} EO autorizadas`:'Seleccionar EO';
    detalle.append(resumen);detalle.open=desplegados.has(c._id);
    detalle.addEventListener('toggle',()=>{if(detalle.open)desplegados.add(c._id);else desplegados.delete(c._id);});
    const aviso=document.createElement('p');aviso.className='ayuda';aviso.textContent='Pueden mostrar el canal; no editar su contenido.';detalle.append(aviso);
    const buscador=document.createElement('input');buscador.type='search';buscador.placeholder='Buscar EO…';buscador.setAttribute('aria-label',`Buscar EO para ${c.codigo}`);buscador.value=filtrosEO.get(c._id)||'';detalle.append(buscador);
    const opciones=document.createElement('div');opciones.className='opciones';detalle.append(opciones);
    const candidatas=eos.filter(e=>e.codigoContexto!==c.enteOperador);
    ids.filter(id=>!eos.some(e=>e.codigoContexto===id)).forEach(id=>candidatas.push({codigoContexto:id,nombre:id+' · no disponible',codigoEO:''}));
    function dibujarOpciones(){
      opciones.replaceChildren();const filtroEO=buscador.value.toLocaleLowerCase('es');
      candidatas.filter(e=>`${e.codigoEO} ${e.nombre}`.toLocaleLowerCase('es').includes(filtroEO)).forEach(e=>{
        const label=document.createElement('label'),check=document.createElement('input'),nombre=document.createElement('span');
        check.type='checkbox';check.checked=ids.includes(e.codigoContexto);check.disabled=ocupado||!conectado;check.dataset.control='distribucion';
        nombre.textContent=[e.codigoEO,e.nombre].filter(Boolean).join(' · ');
        check.addEventListener('change',()=>guardar(c,{eoCompartidas:check.checked?[...ids,e.codigoContexto]:ids.filter(id=>id!==e.codigoContexto)}));
        label.append(check,nombre);opciones.append(label);
      });
      if(!opciones.childNodes.length){const vacio=document.createElement('p');vacio.className='ayuda';vacio.textContent='No hay EO disponibles con este criterio.';opciones.append(vacio);}
    }
    buscador.addEventListener('input',()=>{filtrosEO.set(c._id,buscador.value);dibujarOpciones();});dibujarOpciones();
    compartidas.append(detalle);tr.append(compartidas);
    const hub=document.createElement('td'),labelHub=document.createElement('label'),checkHub=document.createElement('input'),textoHub=document.createElement('span');
    labelHub.className='hub';checkHub.type='checkbox';checkHub.dataset.control='distribucion';checkHub.checked=c.publicarHub===true;checkHub.disabled=ocupado||!conectado;
    checkHub.setAttribute('aria-label',`Publicar ${c.codigo} en TVD HUB`);textoHub.textContent='Autorizar';
    checkHub.addEventListener('change',()=>guardar(c,{publicarHub:checkHub.checked}));labelHub.append(checkHub,textoHub);hub.append(labelHub);tr.append(hub);
    const insignia=document.createElement('td'),labelInsignia=document.createElement('label'),checkInsignia=document.createElement('input'),textoInsignia=document.createElement('span');
    labelInsignia.className='hub';checkInsignia.type='checkbox';checkInsignia.dataset.control='distribucion';checkInsignia.checked=c.esInsigniaEO===true;checkInsignia.disabled=ocupado||!conectado;
    checkInsignia.setAttribute('aria-label',`Canal insignia ${c.codigo}`);textoInsignia.textContent='Insignia';
    checkInsignia.addEventListener('change',()=>guardar(c,{esInsigniaEO:checkInsignia.checked}));labelInsignia.append(checkInsignia,textoInsignia);insignia.append(labelInsignia);tr.append(insignia);
    $('filas').append(tr);
  });
  $('conteo').textContent=recibido?`${visibles.length} de ${canales.length} canales`:'Sin datos recibidos';
  actualizarControles();
}
function guardar(c,cambios){
  if(ocupado||!conectado)return;
  ocupado=true;actualizarControles();
  $('estado').className='';$('estado').textContent=`Guardando configuración de ${c.codigo}…`;
  enviar('TVD_GC_GUARDAR_DISTRIBUCION',{canalId:c._id,enteOperador:c.enteOperador||'',eoCompartidas:c.eoCompartidas||[],publicarHub:c.publicarHub===true,esInsigniaEO:c.esInsigniaEO===true,...cambios});
}
function actualizarControles(){
  $('recargar').disabled=ocupado||!conectado;
  document.querySelectorAll('select,input[data-control]').forEach(s=>s.disabled=ocupado||!conectado);
}
window.addEventListener('message',event=>{
  const m=event.data;
  if(event.source!==parent||!m||m.tipo!=='TVD_GC_ESTADO'||m.version!==VERSION)return;
  conectado=true;clearInterval(reintento);
  if(Array.isArray(m.suscripciones)){
    const zona=$('suscripciones');zona.replaceChildren();
    const titulo=document.createElement('strong');titulo.textContent='Suscripción · TV Digital';zona.append(titulo);
    const etiquetas={VIGENTE:'Vigente',SIN_COMPRA:'Sin suscripción',POR_VINCULAR:'Compra disponible para vincular',ERROR_VERIFICACION:'No se pudo verificar',PAUSED:'Pausada',CANCELED:'Cancelada',ENDED:'Finalizada'};
    m.suscripciones.forEach(s=>{
      const fila=document.createElement('div');fila.style.cssText='display:flex;align-items:center;gap:10px;flex-wrap:wrap;margin:8px 0';
      const texto=document.createElement('span');texto.textContent=(s.nombre||s.codigoEO)+' · '+(etiquetas[s.estado]||s.estado);fila.append(texto);
      (s.comprasDisponibles||[]).forEach((o,i)=>{const b=document.createElement('button');b.type='button';b.dataset.control='suscripcion';b.disabled=['CARGANDO','GUARDANDO'].includes(m.estado);b.textContent='Vincular compra'+(s.comprasDisponibles.length>1?' '+(i+1):'');b.title=o.id;b.onclick=()=>{if(ocupado)return;ocupado=true;actualizarControles();enviar('TVD_GC_VINCULAR_SUSCRIPCION',{codigoEO:s.codigoEO,orderId:o.id});};fila.append(b);});
      zona.append(fila);
    });
  }
  ocupado=['CARGANDO','GUARDANDO'].includes(m.estado);
  $('estado').className=m.estado==='ERROR'?'error':m.estado==='OK'?'ok':'';
  $('estado').textContent=m.mensaje||'';
  if(m.estado==='OK'||m.estado==='ERROR'){
    if(Array.isArray(m.canales))canales=m.canales;
    if(Array.isArray(m.eos))eos=m.eos;
    if(m.estado==='OK')recibido=true;
    render();
  }else actualizarControles();
});
$('buscar').addEventListener('input',render);
$('limpiar').addEventListener('click',()=>{$('buscar').value='';render();$('buscar').focus();});
$('recargar').addEventListener('click',()=>{if(ocupado)return;ocupado=true;actualizarControles();$('estado').textContent='Consultando catálogo…';enviar('TVD_GC_CONSULTAR');});
const reintento=setInterval(()=>{if(++intentos>=10){clearInterval(reintento);if(!conectado)$('estado').textContent='No se recibió conexión. Verifica el Velo de la página y vuelve a cargar.';return;}enviar('TVD_GC_LISTO');},1000);
render();enviar('TVD_GC_LISTO');
</script></body></html>
