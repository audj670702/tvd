async function administradorActual() {

  }

    .query(COLECCION_ADMIN)

function fecha(valor) {
if (!valor) return null;
const f = new Date(valor);
return Number.isNaN(f.getTime()) ? null : f;
}

async function listarCompleto(consulta) {
  let pagina = await consulta.limit(100).find(OPC);
  const items = [...pagina.items];
  while (pagina.hasNext()) { pagina = await pagina.next(); items.push(...pagina.items); }
  return items;
}

function configuracionDistribucion(c = {}) {
  return { enteOperador:typeof c.enteOperador === 'string' ? c.enteOperador : '',
    eoCompartidas:Array.isArray(c.eoCompartidas) ? [...new Set(c.eoCompartidas.filter(e => typeof e === 'string' && e))].sort() : [],
    publicarHub:c.publicarHub === true, esInsigniaEO:c.esInsigniaEO === true };
}
function validarConfiguracion(c) {
  if(c && c.esInsigniaEO !== undefined && typeof c.esInsigniaEO !== 'boolean') throw new Error('INSIGNIA_INVALIDA');
  if (!c || typeof c.enteOperador !== 'string' || !Array.isArray(c.eoCompartidas) ||
    c.eoCompartidas.some(e => typeof e !== 'string' || !e.trim() || e.length > 200) ||
    typeof c.publicarHub !== 'boolean' || c.enteOperador.length > 200) throw new Error('CONFIGURACION_INVALIDA');
  return configuracionDistribucion({ enteOperador:c.enteOperador.trim(),
    eoCompartidas:c.eoCompartidas.map(e => e.trim()), publicarHub:c.publicarHub,esInsigniaEO:c.esInsigniaEO === true });
}
async function guardarDistribucionCanal(datos, admin) {
  const codigo = normalizarCodigo(datos.codigo);
  const id = texto(datos._id,80);
  if (!/^CH\d{2}$/.test(codigo) || !id) throw new Error('CANAL_REQUERIDO');
  const anterior = validarConfiguracion(datos.estadoAnterior);
  const nueva = validarConfiguracion(datos);
  nueva.eoCompartidas = nueva.eoCompartidas.filter(e => e !== nueva.enteOperador);
  const actual = await wixData.get(COL.CANALES,id,OPC);
  if (!actual || actual.codigo !== codigo) throw new Error('CANAL_NO_ENCONTRADO');
  const guardada = configuracionDistribucion(actual);
  if (JSON.stringify(guardada) !== JSON.stringify(anterior)) throw new Error('CONFIGURACION_MODIFICADA_RECARGA');
  const eos = await listarCompleto(wixData.query(COLECCION_EO).eq('app',APP_TVD).eq('activo',true));
  const disponibles = new Set(eos.map(e => e.codigoContexto).filter(Boolean));
  const comprobar = [nueva.enteOperador,...nueva.eoCompartidas].filter(Boolean);
  for (const contexto of comprobar) {
    const conservado = contexto === nueva.enteOperador
      ? contexto === guardada.enteOperador
      : guardada.eoCompartidas.includes(contexto);
    if (!disponibles.has(contexto) && !conservado) throw new Error('EO_NO_DISPONIBLE');
  }
  if (nueva.esInsigniaEO) {
    if (!nueva.enteOperador) throw new Error('INSIGNIA_REQUIERE_EO_PROPIETARIA');
    if (actual.activo !== true) throw new Error('INSIGNIA_REQUIERE_CANAL_ACTIVO');
    const otros=await listarCompleto(wixData.query(COL.CANALES).eq('enteOperador',nueva.enteOperador));
    if (otros.some(c => c._id !== actual._id && c.esInsigniaEO === true)) throw new Error('EO_YA_TIENE_INSIGNIA_DESMARCA_LA_ANTERIOR');
  }
  if (JSON.stringify(guardada) === JSON.stringify(nueva)) return { ok:true, canal:resumenCanalGestion(actual) };
  const resultado = await wixData.update(COL.CANALES,{ ...actual, ...nueva },ESCRITURA);
  await registrar(admin,'CANAL_DISTRIBUCION_GUARDADA',codigo,
    JSON.stringify({ anterior:guardada, nueva }));
  return { ok:true, canal:resumenCanalGestion(resultado) };
}
async function asignarEOCanal(datos, admin) {
  const codigo = normalizarCodigo(datos.codigo);
  const id = texto(datos._id,80);
  if (!/^CH\d{2}$/.test(codigo) || !id) throw new Error('CANAL_REQUERIDO');
  if (typeof datos.enteOperador !== 'string' || typeof datos.enteOperadorAnterior !== 'string') throw new Error('DATOS_INCOMPLETOS');
  const contexto = datos.enteOperador.trim();
  if (contexto.length > 200) throw new Error('EO_INVALIDA');
  if (contexto) {
    const eos = await wixData.query(COLECCION_EO).eq('codigoContexto',contexto).eq('app',APP_TVD).eq('activo',true).limit(2).find(OPC);
    if (eos.items.length !== 1) throw new Error('EO_NO_DISPONIBLE');
  }
  const actual = await wixData.get(COL.CANALES,id,OPC);
  if (!actual || actual.codigo !== codigo) throw new Error('CANAL_NO_ENCONTRADO');
  const anterior = actual.enteOperador || '';
  if (anterior !== datos.enteOperadorAnterior) throw new Error('ASIGNACION_MODIFICADA_RECARGA');
  if (anterior === contexto) return { ok:true, canal:resumenCanalGestion(actual) };
  const guardado = await wixData.update(COL.CANALES,{ ...actual, enteOperador:contexto, esInsigniaEO:anterior === contexto && actual.esInsigniaEO === true, eoCompartidas:configuracionDistribucion(actual).eoCompartidas.filter(e => e !== contexto) },ESCRITURA);
  await registrar(admin,'CANAL_EO_ASIGNADA',codigo,`${anterior || 'SIN EO'} → ${contexto || 'SIN EO'}`);
  return { ok:true, canal:resumenCanalGestion(guardado) };
}

// La parrilla es una lista ordenada (campo `orden`). Inicio y fin se calculan; no se capturan.
// Sólo "Transmitir ahora" corta la señal; agregar, reordenar o activar es programación.
async function filasOrdenadas(cod) {
const r = await wixData.query(COL.PARRILLA).eq('canalId', cod)
    .ascending('orden').ascending('_createdDate').limit(1000).find(OPC);
return r.items;
}
/** Renumera `orden` 1..n según la lista de registros recibida (sólo escribe los que cambian). */
async function renumerar(registros) {
let orden = 1;
for (const e of registros) {
if (Number(e.orden) !== orden) {
await wixData.update(COL.PARRILLA, { ...e, orden }, ESCRITURA);
    }
    orden += 1;
  }
}

/**
 * Agrega o edita una fila.
 * posicion: 'FINAL' (predeterminado al agregar), 'ACTUAL' (al editar) o un número 1..n.
 */
async function op_tvdGuardarFila(admin, datos = {}) {
const cod = normalizarCodigo(datos.canalId);
const tipo = texto(datos.tipoContenido, 40).toUpperCase();
if (!cod) throw new Error('CANAL_REQUERIDO');
if (!TIPOS_FILA.includes(tipo)) throw new Error('TIPO_INVALIDO');
const canal = await wixData.query(COL.CANALES).eq('codigo', cod).limit(1).find(OPC);
if (!canal.items.length) throw new Error('CANAL_NO_ENCONTRADO');
const registro = {
    canalId: cod,
    tipoContenido: tipo,
    titulo: texto(datos.titulo, 180),
    descripcion: texto(datos.descripcion, 1000),
    publicado: datos.publicado !== false,
    contenidoId: '',
    catalogoId: '',
    url: '',
    tipoFuente: '',
    duracionSegundos: 0
  };
if (tipo === 'PLAYLIST') {
const playlistId = await resolverPlaylistId(datos.contenidoId);
if (!playlistId) throw new Error('PLAYLIST_REQUERIDA');
if ((await duracionPlaylistActiva(playlistId)) <= 0) throw new Error('PLAYLIST_SIN_DURACION');
    registro.contenidoId = playlistId;
if (!registro.titulo) {
const p = await wixData.get(COL.PLAYLISTS, playlistId, OPC);
      registro.titulo = texto(p?.nombre, 180);
    }
  } else {
const catalogoId = texto(datos.catalogoId, 80);
let url = texto(datos.url, 1500);
let duracion = segundos(datos.duracionSegundos);
if (catalogoId) {
const c = await wixData.get(COL.CATALOGO, catalogoId, OPC);
if (!c) throw new Error('CATALOGO_NO_ENCONTRADO');
if (!url && !texto(c.urlYoutube, 1500)) throw new Error('CATALOGO_SIN_YOUTUBE');
      url = url || texto(c.urlYoutube, 1500);
      duracion = duracion || segundos(c.duracion);
      registro.titulo = registro.titulo || texto(c.nombreVideo || c.titulo, 180);
      registro.catalogoId = catalogoId;
    }
if (!url) throw new Error('URL_REQUERIDA');
if (inferirTipoFuente(url) !== 'YOUTUBE') throw new Error('URL_NO_YOUTUBE');
if (duracion <= 0) throw new Error('DURACION_REQUERIDA');
    registro.url = url;
    registro.tipoFuente = 'YOUTUBE';
    registro.duracionSegundos = duracion;
  }
if (!registro.titulo) throw new Error('TITULO_REQUERIDO');
const id = texto(datos._id, 80);
const lista = await filasOrdenadas(cod);
let guardado;
let resto = lista;
if (id) {
const actual = lista.find((e) => e._id === id);
if (!actual) throw new Error('FILA_NO_ENCONTRADA');
    guardado = await wixData.update(COL.PARRILLA, { ...actual, ...registro }, ESCRITURA);
    resto = lista.filter((e) => e._id !== id);
const pos = datos.posicion === 'ACTUAL' || datos.posicion === undefined
      ? lista.findIndex((e) => e._id === id)
      : (datos.posicion === 'FINAL' ? resto.length : Math.min(resto.length, Math.max(0, Number(datos.posicion) - 1)));
    resto.splice(pos, 0, guardado);
  } else {
    guardado = await wixData.insert(COL.PARRILLA, { ...registro, orden: lista.length + 1 }, ESCRITURA);
const pos = !datos.posicion || datos.posicion === 'FINAL'
      ? resto.length
      : Math.min(resto.length, Math.max(0, Number(datos.posicion) - 1));
    resto = [...resto];
    resto.splice(pos, 0, guardado);
  }
await renumerar(resto);
await registrar(admin, id ? 'FILA_EDITADA' : 'FILA_AGREGADA', cod, registro.titulo);
return { ok: true, fila: guardado };
}
export const tvdGuardarFila = webMethod(Permissions.SiteMember, async (datos = {}) => op_tvdGuardarFila(await exigirAdministrador(), datos));
/** Nuevo orden de la parrilla: recibe la lista de _id en el orden deseado. No corta la señal. */
async function op_tvdReordenarFilas(admin, canalId, ids = []) {
const cod = normalizarCodigo(canalId);
const lista = await filasOrdenadas(cod);
const porId = new Map(lista.map((e) => [e._id, e]));
const nuevo = (Array.isArray(ids) ? ids : []).map((x) => porId.get(x)).filter(Boolean);
  lista.forEach((e) => { if (!nuevo.includes(e)) nuevo.push(e); });
await renumerar(nuevo);
await registrar(admin, 'PARRILLA_REORDENADA', cod, '');
return { ok: true };
}

/** "Transmitir ahora": corte inmediato; la fila arranca ahora y el ciclo se recalcula desde ella. */
async function op_tvdTransmitirAhora(admin, canalId, filaId) {
const cod = normalizarCodigo(canalId);
const fila = await wixData.get(COL.PARRILLA, texto(filaId, 80), OPC);
if (!fila || fila.canalId !== cod) throw new Error('FILA_NO_ENCONTRADA');
if (fila.publicado === false) throw new Error('FILA_INACTIVA');
await guardarEstado(cod, {
    anclaEntradaId: fila._id,
    anclaDesde: new Date(),
    manualEntradaId: '',
    manualDesde: null
  }, admin);
await registrar(admin, 'TRANSMITIR_AHORA', cod, fila.titulo);
return { ok: true };
}

function segundosDesdeTexto(valor) {
if (typeof valor === 'number') return Math.max(0, Math.floor(valor));
const partes = String(valor || '').trim().split(':').map(Number);
if (!partes.length || partes.some((n) => Number.isNaN(n))) return 0;
if (partes.length === 3) return partes[0] * 3600 + partes[1] * 60 + partes[2];
if (partes.length === 2) return partes[0] * 60 + partes[1];
return Math.max(0, partes[0] || 0);
}
